INSERT INTO registro_vacinacao
(
    id_animal,
    id_vacina,
    id_clinica,
    id_veterinario,
    data_aplicacao,
    data_proxima_dose,
    status
)
SELECT
    a.id_animal,
    ((a.id_animal - 1) MOD 20) + 1,
    ((a.id_animal - 1) MOD 11) + 1,
    ((a.id_animal - 1) MOD 26) + 1,
    DATE_ADD('2024-01-01', INTERVAL (a.id_animal * 5) DAY),
    DATE_ADD('2025-01-01', INTERVAL (a.id_animal * 5) DAY),
    CASE
        WHEN a.id_animal % 10 = 0 THEN 'CANCELADA'
        WHEN a.id_animal % 5 = 0 THEN 'ATRASADA'
        WHEN a.id_animal % 3 = 0 THEN 'PENDENTE'
        ELSE 'APLICADA'
    END
FROM animal a;

-- 09_LOG_AUDITORIA.SQL (PARTE 2)
-- Massa de Dados Homologação - ImunoPet
-- =====================================================


INSERT INTO log_auditoria
(
    id_usuario,
    acao,
    detalhes,
    data_hora
)
SELECT
    u.id_usuario,
    CASE
        WHEN u.id_usuario % 6 = 0 THEN 'LOGIN'
        WHEN u.id_usuario % 6 = 1 THEN 'CONSULTAR_ANIMAL'
        WHEN u.id_usuario % 6 = 2 THEN 'CONSULTAR_HISTORICO'
        WHEN u.id_usuario % 6 = 3 THEN 'REGISTRAR_VACINACAO'
        WHEN u.id_usuario % 6 = 4 THEN 'GERAR_RELATORIO'
        ELSE 'LOGOUT'
    END,
    CONCAT('Ação automática de homologação para usuário ', u.id_usuario),
    DATE_ADD('2026-01-05 08:00:00', INTERVAL u.id_usuario HOUR)
FROM usuario u;
